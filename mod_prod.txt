' Cancel

code = code & "End Sub"

vbComp.CodeModule.AddFromString code

MsgBox "Productivity Form Updated (Aligned Bottoms, Compact Height).", vbInformation

' ==========================================================================================
' MODULE: modProductivity (V11: FINAL - DECOUPLED CHARTS)
' Features:
'   - KPIs/Forecast tied to User Input Range
'   - Monthly Chart FIXED to 4 Months ending on EndDate
'   - Daily Chart FIXED to 7 Days ending on EndDate
'   - Shape Button with Custom Styling
' ==========================================================================================
Option Explicit

Sub ShowProductivityForm()
On Error Resume Next
ufProductivityParams.Show
End Sub

Sub GenerateProductivityReport(sDate As Date, eDate As Date, targetAmt As Double)
Dim wsDockets As Worksheet
Dim wbNew As Workbook, wsOut As Worksheet
Dim cell As Range
Dim dictClients As Object, dictMonths As Object, dictDaily As Object

' --- 1. CONFIGURATION ---
Const D_COL_DATE = 1
Const D_COL_CLIENT = 2
Const D_COL_TIME = 5
Const D_COL_RATE = 6
Const D_COL_PCT = 7

' Styles
Dim colDesk As Long: colDesk = RGB(230, 230, 230)
Dim colPaper As Long: colPaper = vbWhite
Dim colHeader As Long: colHeader = RGB(30, 40, 60)
Dim colTextMain As Long: colTextMain = RGB(40, 40, 40)
Dim colTextMuted As Long: colTextMuted = RGB(110, 110, 110)
Dim colBrand As Long: colBrand = RGB(52, 152, 219)

Set wsDockets = ThisWorkbook.Sheets("Dockets")
Set dictClients = CreateObject("Scripting.Dictionary")
Set dictMonths = CreateObject("Scripting.Dictionary")
Set dictDaily = CreateObject("Scripting.Dictionary")

Application.ScreenUpdating = False

' --- 2. DEFINE DATE RANGES ---
' A. User Range (for KPIs & Forecast) -> sDate to eDate

' B. Monthly Chart Range (Fixed: 4 Months ending at eDate)
Dim dChartMonthStart As Date
dChartMonthStart = DateSerial(Year(eDate), Month(eDate) - 3, 1) ' Current month - 3 prior

' C. Daily Chart Range (Fixed: 7 Days ending at eDate)
Dim dChartDailyStart As Date
dChartDailyStart = eDate - 6

' D. Data Fetch Start (The earliest of all starts)
Dim dFetchStart As Date
dFetchStart = sDate
If dChartMonthStart < dFetchStart Then dFetchStart = dChartMonthStart
If dChartDailyStart < dFetchStart Then dFetchStart = dChartDailyStart

' --- 3. PRE-FILL CHART DICTIONARIES ---
' This ensures zero-value periods appear in the charts
Dim i As Long, dRunner As Date

' Pre-fill Daily (Last 7 Days)
For i = 6 To 0 Step -1
dictDaily(eDate - i) = 0#
Next i

' Pre-fill Monthly (Last 4 Months)
For i = 0 To 3
dRunner = DateSerial(Year(eDate), Month(eDate) - i, 1)
dictMonths(dRunner) = 0#
Next i

' --- 4. GATHER DATA ---
Dim dDate As Date, dClient As String
Dim dTime As Double, dRate As Double, dPct As Double, dFee As Double
Dim totalHours As Double, totalFees As Double
Dim mKey As Date

If wsDockets.Cells(wsDockets.rows.count, 1).End(xlUp).row > 1 Then
For Each cell In wsDockets.Range(wsDockets.Cells(2, 1), wsDockets.Cells(wsDockets.rows.count, 1).End(xlUp))
If IsDate(cell.value) Then
dDate = CDate(cell.value)

' Only process if within the widest possible range we need
If dDate >= dFetchStart And dDate <= eDate Then

' Read Values
dClient = Trim(UCase(cell.Offset(0, D_COL_CLIENT - 1).value))
dTime = val(cell.Offset(0, D_COL_TIME - 1).value)
dRate = val(cell.Offset(0, D_COL_RATE - 1).value)
dPct = val(cell.Offset(0, D_COL_PCT - 1).value)
If cell.Offset(0, D_COL_PCT - 1).value = "" Then dPct = 1
dFee = dTime * (dRate * dPct)

' A. KPI AGGREGATION (User Range Only)
If dDate >= sDate Then
totalHours = totalHours + dTime
totalFees = totalFees + dFee

' Client List (User Range Only)
If dictClients.Exists(dClient) Then
dictClients(dClient) = dictClients(dClient) + dFee
Else
dictClients(dClient) = dFee
End If
End If

' B. MONTHLY CHART AGGREGATION (Fixed Range)
If dDate >= dChartMonthStart Then
mKey = DateSerial(Year(dDate), Month(dDate), 1)
If dictMonths.Exists(mKey) Then
dictMonths(mKey) = dictMonths(mKey) + dFee
End If
End If

' C. DAILY CHART AGGREGATION (Fixed Range)
If dDate >= dChartDailyStart Then
If dictDaily.Exists(dDate) Then
dictDaily(dDate) = dictDaily(dDate) + dFee
End If
End If

End If
End If
Next cell
End If

' --- 5. CREATE SHEET & SETUP ---
Set wbNew = Workbooks.Add
Set wsOut = wbNew.Sheets(1)
wsOut.name = "Productivity"
wsOut.DisplayPageBreaks = False

With ActiveWindow
.DisplayGridlines = False
.DisplayHeadings = False
.Zoom = 75
End With

' Backgrounds
wsOut.Cells.Interior.color = colDesk
wsOut.Cells.Font.name = "Segoe UI"

' The Paper
With wsOut.Range("B2:M48")
.Interior.color = colPaper
End With

' Store Dates for PDF macro
wsOut.Cells(1, 26).value = sDate
wsOut.Cells(2, 26).value = eDate

' Layout Columns
wsOut.Columns("A").ColumnWidth = 4
wsOut.Columns("B").ColumnWidth = 2
wsOut.Columns("C:E").ColumnWidth = 12
wsOut.Columns("F").ColumnWidth = 2
wsOut.Columns("G:H").ColumnWidth = 12
wsOut.Columns("I").ColumnWidth = 2
wsOut.Columns("J").ColumnWidth = 25
wsOut.Columns("K:L").ColumnWidth = 10
wsOut.Columns("M").ColumnWidth = 2
wsOut.Columns("N").ColumnWidth = 22

' Header Bar
With wsOut.Range("B2:M5")
.Interior.color = colHeader
End With

With wsOut
.Cells(3, 3).value = "PRODUCTIVITY REPORT"
.Cells(4, 3).value = "Cory Schneider Law Office"
.Cells(3, 3).Font.Size = 20: .Cells(3, 3).Font.Bold = True: .Cells(3, 3).Font.color = vbWhite
.Cells(4, 3).Font.Size = 10: .Cells(4, 3).Font.color = RGB(200, 200, 200)

.Range("J3:L4").Merge
.Range("J3").value = Format(sDate, "MMM d, yyyy") & " - " & Format(eDate, "MMM d, yyyy")
.Range("J3").Font.color = vbWhite: .Range("J3").HorizontalAlignment = xlRight
.Range("J3").VerticalAlignment = xlCenter
.Range("J3").Font.Size = 11
End With

' --- 6. KPIs (User Range) ---
Dim realizedRate As Double
If totalHours > 0 Then realizedRate = totalFees / totalHours Else realizedRate = 0
DrawCleanMetric wsOut, wsOut.Range("C7:E9"), "TOTAL PRODUCTION", totalFees, "$#,##0", colTextMain
DrawCleanMetric wsOut, wsOut.Range("G7:H9"), "BILLABLE HOURS", totalHours, "#,##0.0", colTextMain
DrawCleanMetric wsOut, wsOut.Range("J7:L9"), "REALIZED RATE", realizedRate, "$#,##0.00", colTextMain

' --- 7. FORECAST (User Range) ---
Dim daysInPeriod As Long, dailyAvg As Double, projectedAnnual As Double, pacePct As Double
daysInPeriod = DateDiff("d", sDate, eDate) + 1
If daysInPeriod < 1 Then daysInPeriod = 1
dailyAvg = totalFees / daysInPeriod
projectedAnnual = dailyAvg * 336
If targetAmt > 0 Then pacePct = (projectedAnnual / targetAmt) - 1 Else pacePct = 0

With wsOut.Cells(12, 3)
.value = "ANNUAL FORECAST"
.Font.Size = 11: .Font.Bold = True: .Font.color = colBrand
End With
DrawForecastRow wsOut, 14, "Current Pace", Format(dailyAvg, "$#,##0") & " / day"
DrawForecastRow wsOut, 15, "Basis", "336 Days (48 wks)"
DrawForecastRow wsOut, 16, "Target", Format(targetAmt, "$#,##0")

With wsOut.Range("C18:H19")
.Merge: .value = projectedAnnual: .NumberFormat = "$#,##0"
.Font.name = "Segoe UI Light": .Font.Size = 32: .Font.color = colTextMain
.HorizontalAlignment = xlLeft: .VerticalAlignment = xlCenter
End With

Dim statusText As String, statusColor As Long
If pacePct >= 0 Then
statusText = "+" & Format(pacePct, "0.0%") & " vs Target"
statusColor = RGB(39, 174, 96)
Else
statusText = Format(pacePct, "0.0%") & " vs Target"
statusColor = RGB(192, 57, 43)
End If
With wsOut.Range("C20:H20")
.Merge: .value = statusText: .Font.color = statusColor
.Font.Bold = True: .Font.Size = 10: .HorizontalAlignment = xlLeft
End With

' --- 8. TOP CLIENTS (User Range) ---
With wsOut.Cells(12, 10)
.value = "TOP CLIENTS"
.Font.Size = 11: .Font.Bold = True: .Font.color = colBrand
End With

Dim rCli As Long: rCli = 2
wsOut.Cells(1, 27).value = "C": wsOut.Cells(1, 28).value = "V"
Dim k As Variant
For Each k In dictClients.Keys
wsOut.Cells(rCli, 27).value = k: wsOut.Cells(rCli, 28).value = dictClients(k)
rCli = rCli + 1
Next k
If rCli > 2 Then
wsOut.Range("AA1:AB" & rCli - 1).Sort Key1:=wsOut.Range("AB1"), Order1:=xlDescending, Header:=xlYes
End If
Dim X As Integer
For X = 0 To 4
If X + 2 < rCli Then
wsOut.Cells(14 + X, 10).value = wsOut.Cells(2 + X, 27).value
wsOut.Cells(14 + X, 10).Font.Size = 9: wsOut.Cells(14 + X, 10).Font.color = colTextMain
wsOut.Cells(14 + X, 11).value = wsOut.Cells(2 + X, 28).value
wsOut.Cells(14 + X, 11).NumberFormat = "$#,##0": wsOut.Cells(14 + X, 11).Font.Size = 9
wsOut.Cells(14 + X, 11).HorizontalAlignment = xlRight
wsOut.Cells(14 + X, 12).value = wsOut.Cells(2 + X, 28).value
wsOut.Cells(14 + X, 12).Font.color = colPaper
End If
Next X
With wsOut.Range("L14:L18").FormatConditions.AddDatabar
.BarColor.color = colBrand: .BarFillType = xlDataBarFillSolid: .ShowValue = False
End With

' --- 9. CHARTS (Fixed Ranges) ---

' A. Monthly Chart (4 Months ending at eDate)
Dim rChart As Long: rChart = 2
wsOut.Cells(1, 31).value = "Month": wsOut.Cells(1, 32).value = "Fees"
For Each k In dictMonths.Keys
wsOut.Cells(rChart, 31).value = k: wsOut.Cells(rChart, 32).value = dictMonths(k)
rChart = rChart + 1
Next k
If rChart > 2 Then
wsOut.Range("AE1:AF" & rChart - 1).Sort Key1:=wsOut.Range("AE1"), Order1:=xlAscending, Header:=xlYes
CreateModernChart wsOut, wsOut.Range("C22"), wsOut.Range("AE1:AF" & rChart - 1), "Monthly Production Trend (Last 4 Months)", colTextMuted, colHeader, 175
End If

' B. Daily Chart (7 Days ending at eDate)

' Reusable Chart Creator
Sub CreateModernChart(ws As Worksheet, anchor As Range, sourceRng As Range, Title As String, colTitle As Long, colBar As Long, h As Double)
Dim chtObj As ChartObject
Set chtObj = ws.ChartObjects.Add(Left:=anchor.Left, Top:=anchor.Top, _
Width:=ws.Range("C" & anchor.row & ":L" & anchor.row).Width, Height:=h)
With chtObj.Chart
.ChartType = xlColumnClustered
.SetSourceData Source:=sourceRng
.PlotVisibleOnly = False
.HasTitle = True
.ChartTitle.Text = Title
.ChartTitle.Font.name = "Segoe UI"
.ChartTitle.Font.Size = 10
.ChartTitle.Font.color = colTitle
.Legend.Delete
With .SeriesCollection(1)
.Interior.color = colBar
.ApplyDataLabels
.DataLabels.position = xlLabelPositionOutsideEnd
.DataLabels.NumberFormat = "$#,##0"
.DataLabels.Font.Size = 8
.DataLabels.Font.name = "Segoe UI"
End With
.Axes(xlValue).Delete
.Axes(xlValue).MajorGridlines.Delete
With .Axes(xlCategory).TickLabels
.Font.name = "Segoe UI"
.Font.Size = 8
.NumberFormat = "d-mmm"
End With
.ChartArea.Format.line.Visible = msoFalse
.PlotArea.Format.line.Visible = msoFalse
.ChartArea.Interior.color = vbWhite
.PlotArea.Interior.color = vbWhite
End With
End Sub

Sub SaveProductivityToPDF()
Dim ws As Worksheet
Dim savePath As String, fileName As String, rootPath As String
Dim sDate As Date, eDate As Date
Set ws = ActiveSheet

ws.DisplayPageBreaks = False

On Error Resume Next
sDate = CDate(ws.Cells(1, 26).value)
eDate = CDate(ws.Cells(2, 26).value)
On Error GoTo 0
rootPath = M6_GetUnifiedBaseFolder()
If Right(rootPath, 1) <> "\" Then rootPath = rootPath & "\"
savePath = rootPath & "__PRODUCTIVITY\"
If Dir(savePath, vbDirectory) = "" Then
On Error Resume Next: MkDir savePath: On Error GoTo 0
End If
fileName = "Productivity " & Format(sDate, "yy-mm-dd") & " to " & Format(eDate, "yy-mm-dd") & ".pdf"

Application.ScreenUpdating = False

With ws.PageSetup
.Zoom = False
.FitToPagesWide = 1
.FitToPagesTall = 1
End With

On Error Resume Next
ws.ExportAsFixedFormat Type:=xlTypePDF, fileName:=savePath & fileName, _
Quality:=xlQualityStandard, IncludeDocProperties:=True, IgnorePrintAreas:=False, OpenAfterPublish:=True
If err.Number <> 0 Then MsgBox "Error saving PDF: " & err.description, vbCritical
On Error GoTo 0

ws.DisplayPageBreaks = False
Application.ScreenUpdating = True
End Sub
